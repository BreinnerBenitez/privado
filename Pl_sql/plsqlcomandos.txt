SET SERVEROUTPUT ON;

DECLARE
    --  Definición del Cursor Explícito
    CURSOR c_empleados IS 
        SELECT nombre, salario, dias_trabajados 
        FROM empleados;

   -- Procedimiento de impresión recibe un registro como parámetro
    PROCEDURE imprimir_empleado(p_nombre VARCHAR2, p_salario NUMBER, p_dias NUMBER) IS
        v_salario_diario NUMBER;
        v_salario_pagar  NUMBER;
    BEGIN
        v_salario_diario := p_salario / 30;
        v_salario_pagar  := v_salario_diario * p_dias;

        DBMS_OUTPUT.PUT_LINE('========================');
        DBMS_OUTPUT.PUT_LINE('Empleado: ' || p_nombre);
        -- ROUND para que el salario diario se vea limpio con 2 decimales
        DBMS_OUTPUT.PUT_LINE('Salario diario: ' || ROUND(v_salario_diario, 2));
        DBMS_OUTPUT.PUT_LINE('Dias Trabajados: ' || p_dias);
        DBMS_OUTPUT.PUT_LINE('Salario a Pagar: ' || ROUND(v_salario_pagar, 2));
    END;

BEGIN
    -- 3. Ciclo para recorrer el cursor
    FOR r_emp IN c_empleados LOOP
        imprimir_empleado(r_emp.nombre, r_emp.salario, r_emp.dias_trabajados);
    END LOOP;
    
    DBMS_OUTPUT.PUT_LINE('========================');
END;
/
